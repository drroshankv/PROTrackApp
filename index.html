📋 Google Sheets Apps Script Code (Includes In/Out Times)

1. Open a Google Sheet and go to Extensions > Apps Script

2. Paste this code:

function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(["Date", "Reached Time (In)", "Left Time (Out)", "Duration (Mins)", "Doctor Name", "Met Status", "Facility / Type", "Discussion / Quick Pills / Reason", "Executive", "Employee ID", "Latitude", "Longitude", "Accuracy (m)", "Google Maps URL"]); } var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.date_str || new Date().toISOString().slice(0, 10), data.reached_time || "", data.left_time || "", data.duration_minutes || "", data.doctor_name || "", data.met_status || "", data.facility_name || data.location_type || "", data.notes_or_pitch || data.reason_not_met || "", data.rep_name || "", data.rep_id || "", data.latitude || "", data.longitude || "", data.accuracy_meters || "", data.maps_url || "" ]); return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", error: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }

3. Deploy as Web app (Access: Anyone) and paste URL above.
