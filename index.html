/** * PRO Tracker - Google Sheets Integration * * Doctor Master Sheet: Tab "Doctor List" * Fills doctor details from Column B onwards without skipping columns. * Col A (1): S.No * Col B (2): Doctor Name (Auto-prefixed with "DR ") * Col C (3): Speciality / Department * Col D (4): Clinic / Hospital Name * Col E (5): Contact Number * Col F (6): OP Days & Time * Col G (7): Best Time to Meet * Col H (8): Priority (High / Moderate / Low) * Col I (9): GPS Latitude * Col J (10): GPS Longitude * Col K (11): Location Link (Google Maps) * Col L (12): Assigned PRO (Last Column - Multiple executives separated by comma) * * Visit Call Logs Sheet: Tab "Visit Data" * Captures field visit logs: * Date, Reached Time, Left Time, Duration Minutes, Doctor Name, Department, * Contact Number, Met Status, Facility or Type, Discussion Notes, * Executive Name, Employee ID, Latitude, Longitude, Accuracy Meters, Google Maps URL */ // Configuration var DOCTOR_SHEET_NAME = "Doctor List"; // Tab name for Doctor Directory var CALLS_SHEET_NAME = "Visit Data"; // Tab name for Visit Call Logs var DOCTORS_SPREADSHEET_ID = null; var CALLS_SPREADSHEET_ID = null; function getDoctorSpreadsheet() { if (DOCTORS_SPREADSHEET_ID) { return SpreadsheetApp.openById(DOCTORS_SPREADSHEET_ID); } return SpreadsheetApp.getActiveSpreadsheet(); } function getCallsSpreadsheet() { if (CALLS_SPREADSHEET_ID) { return SpreadsheetApp.openById(CALLS_SPREADSHEET_ID); } return SpreadsheetApp.getActiveSpreadsheet(); } function ensureMinColumns(sheet, minCols) { try { var maxCols = sheet.getMaxColumns(); if (maxCols < minCols) { sheet.insertColumnsAfter(maxCols, minCols - maxCols); } } catch (e) {} } // Format doctor name with "DR " prefix function formatDocNameWithDR(name) { var str = (name || "").toString().trim(); str = str.replace(/^(DR\.?|DOCTOR)\s*/i, "").trim(); return str ? function resolveColMap(docSheet) { var lastCol = Math.max(docSheet.getLastColumn(), 12); var headerRow = []; if (docSheet.getLastRow() >= 1) { headerRow = docSheet.getRange(1, 1, 1, lastCol).getValues()[0]; } else if (hText.indexOf("doctor") !== -1 || hText === "name" || hText === "dr name") { colMap.name = col1Based; } else if (hText.indexOf("special") !== -1 || hText.indexOf("dept") !== -1) { colMap.speciality = col1Based;
